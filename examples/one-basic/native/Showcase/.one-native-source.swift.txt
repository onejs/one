  private struct Request: Decodable { let module: String; let method: String; let argsJson: String; let contractHash: String }
  private struct Reply: Encodable { let ok: Bool; let value: String?; let error: String?; let pending: Int32? }
  private static var output: [UInt8] = []
  private static var finished: [Int32: Reply] = [:]
  private static var nextCall: Int32 = 0
  // peach runs tasks inline, so a call finishes before its task launch returns
  // unless it waits on the clock; then it answers pending and the host polls
  // it after running the timers.
  static func callWasm(_ payload: UnsafePointer<UInt8>?, _ length: Int32) -> UnsafeMutablePointer<UInt8>? {
    let id = nextCall
    nextCall &+= 1
    do {
      guard let payload, length >= 0 else { throw NSError(domain: "OneNativeSource", code: 2, userInfo: [NSLocalizedDescriptionKey: "missing call payload"]) }
      let request = try JSONDecoder().decode(Request.self, from: Data(bytes: payload, count: Int(length)))
      guard request.contractHash == hash else { throw NSError(domain: "OneNativeSource", code: 4, userInfo: [NSLocalizedDescriptionKey: "native module \(request.module) changed; rebuild the app"]) }
      Task {
        do {
          finished[id] = Reply(ok: true, value: try await call(request.module, request.method, request.argsJson), error: nil, pending: nil)
        } catch {
          // the native hosts name the method on everything a call throws
          finished[id] = Reply(ok: false, value: nil, error: "\(request.module).\(request.method): \(error.localizedDescription)", pending: nil)
        }
      }
    } catch {
      finished[id] = Reply(ok: false, value: nil, error: error.localizedDescription, pending: nil)
    }
    return pollWasm(id)
  }
  static func pollWasm(_ id: Int32) -> UnsafeMutablePointer<UInt8>? {
    let reply = finished.removeValue(forKey: id) ?? Reply(ok: true, value: nil, error: nil, pending: id)
    let bytes = (try? JSONEncoder().encode(reply)) ?? Data("{\"ok\":false,\"error\":\"encoding failed\"}".utf8)
    let count = UInt32(bytes.count)
    output = [UInt8(count & 0xff), UInt8((count >> 8) & 0xff), UInt8((count >> 16) & 0xff), UInt8(count >> 24)] + Array(bytes)
    return output.withUnsafeMutableBufferPointer { $0.baseAddress }
  }
  private struct Event: Encodable { let channel: String; let args: String }
  private static var pendingEvents: [Event] = []
  // under wasi there is no javascript runtime, so posted events queue
  // here and the peach host drains them with peach_poll_events, using
  // the same length-prefixed framing as pollWasm.
  static func pollEvents() -> UnsafeMutablePointer<UInt8>? {
    let events = pendingEvents
    pendingEvents = []
    let bytes = (try? JSONEncoder().encode(events)) ?? Data("[]".utf8)
    let count = UInt32(bytes.count)
    output = [UInt8(count & 0xff), UInt8((count >> 8) & 0xff), UInt8((count >> 16) & 0xff), UInt8(count >> 24)] + Array(bytes)
    return output.withUnsafeMutableBufferPointer { $0.baseAddress }
  }

  private static let DeviceInstance = Device()
  private nonisolated static let isoDate = Date.ISO8601FormatStyle()
  private nonisolated static let isoDateFractional = Date.ISO8601FormatStyle(includingFractionalSeconds: true)
  private nonisolated static func decodeDate(_ decoder: Decoder) throws -> Date {
    let container = try decoder.singleValueContainer()
    let text = try container.decode(String.self)
    if let date = (try? isoDateFractional.parse(text)) ?? (try? isoDate.parse(text)) { return date }
    throw DecodingError.dataCorruptedError(in: container, debugDescription: "expected ISO date")
  }
  private static func decode<T: Decodable>(_ type: T.Type, from data: Data) throws -> T {
    let decoder = JSONDecoder()
    decoder.dateDecodingStrategy = .custom { try decodeDate($0) }
    decoder.dataDecodingStrategy = .base64
    return try decoder.decode(T.self, from: data)
  }
  private static func decode<T: Decodable>(_ value: Any, as type: T.Type) throws -> T {
    try decode(T.self, from: try JSONSerialization.data(withJSONObject: value, options: [.fragmentsAllowed]))
  }
  private static func encode(_ value: any Encodable) throws -> String {
    let encoder = JSONEncoder()
    encoder.dateEncodingStrategy = .iso8601
    encoder.dataEncodingStrategy = .base64
    return String(decoding: try encoder.encode(value), as: UTF8.self)
  }
  private static func encodeAny(_ value: any Encodable) -> String {
    (try? encode(value)) ?? "null"
  }
  // subscribe cases and view callbacks both encode callback arguments
  // as a json array, so this helper stays outside the view-only glue.
  private static func args(_ values: [any Encodable]) -> String {
    "[" + values.map { encodeAny($0) }.joined(separator: ",") + "]"
  }

  static func call(_ module: String, _ method: String, _ argsJson: String) async throws -> String {
    guard let values = try JSONSerialization.jsonObject(with: Data(argsJson.utf8)) as? [Any] else {
      throw NSError(domain: "OneNativeSource", code: 2, userInfo: [NSLocalizedDescriptionKey: "arguments must be a JSON array"])
    }
    switch module {
    case "Showcase.Device":
      switch method {
      case "info":
        guard values.count == 0 else { throw NSError(domain: "OneNativeSource", code: 2, userInfo: [NSLocalizedDescriptionKey: "Showcase.Device.info expects 0 arguments"]) }
        let result = DeviceInstance.info()
        return try encode(result)
      case "sha256":
        guard values.count == 1 else { throw NSError(domain: "OneNativeSource", code: 2, userInfo: [NSLocalizedDescriptionKey: "Showcase.Device.sha256 expects 1 arguments"]) }

      default: throw NSError(domain: "OneNativeSource", code: 1, userInfo: [NSLocalizedDescriptionKey: "native method \(module).\(method) is not linked; rebuild the app"])
      }
    default:
      throw NSError(domain: "OneNativeSource", code: 1, userInfo: [NSLocalizedDescriptionKey: "native module \(module) is not linked; rebuild the app"])
    }

  private struct OneListPatch<E: Decodable>: Decodable {
    let length: Int
    let set: [(Int, E)]
    private struct Entry: Decodable {
      let index: Int
      let value: E
      init(from decoder: Decoder) throws {
        var c = try decoder.unkeyedContainer()
        index = try c.decode(Int.self)
        value = try c.decode(E.self)
      }
    }
    private enum CodingKeys: String, CodingKey { case length, set }
    init(from decoder: Decoder) throws {
      let c = try decoder.container(keyedBy: CodingKeys.self)
      length = try c.decode(Int.self, forKey: .length)
      set = try c.decode([Entry].self, forKey: .set).map { ($0.index, $0.value) }
    }
  }
  private nonisolated static func patch<E>(_ list: inout [E], _ patch: OneListPatch<E>) -> Bool {
    if patch.length < 0 { return false }
    if list.count > patch.length { list.removeLast(list.count - patch.length) }
    for (index, value) in patch.set {
      if index < list.count { list[index] = value } else if index == list.count && index < patch.length { list.append(value) } else { return false }
    }
    return list.count == patch.length
  }
  private nonisolated static func missing(_ prop: String) -> Error {
    NSError(domain: "OneNativeSource", code: 1, userInfo: [NSLocalizedDescriptionKey: "missing prop \(prop)"])
  }
  private struct Doc_Level: Decodable {
    let seq: Int
    let base: Int?
    let p0: Optional<Double>
    private enum CodingKeys: String, CodingKey { case seq = "$seq"; case base = "$base"; case lists = "$lists"; case p0 = "value" }
    init(from decoder: Decoder) throws {
      let c = try decoder.container(keyedBy: CodingKeys.self)
      seq = try c.decodeIfPresent(Int.self, forKey: .seq) ?? 0
      base = try c.decodeIfPresent(Int.self, forKey: .base)
      p0 = try c.contains(.p0) ? Optional<Double>.some(c.decode(Double.self, forKey: .p0)) : Optional<Double>.none
    }
  }
  private final class Held_Level {
    var oneSeq: Int
    var `value`: Double
    init(_ d: Doc_Level) throws {
      oneSeq = d.seq
      guard let p0 = d.p0 else { throw OneNativeSourceBody_Showcase.missing("value") }
      `value` = p0
    }
    func apply(_ d: Doc_Level) -> Bool {
      if d.base != oneSeq { return false }
      if let v = d.p0 { `value` = v }
      oneSeq = d.seq
      return true
    }
  }
  static func update(_ held: AnyObject?, _ name: String, _ text: String) throws -> AnyObject? {
    let data = Data(text.utf8)
    switch name {
    case "Level":
      let doc = try decode(Doc_Level.self, from: data)
      if doc.base == nil { return try Held_Level(doc) }
      guard let h = held as? Held_Level, h.apply(doc) else { return nil }
      return h
    default: throw NSError(domain: "OneNativeSource", code: 1, userInfo: [NSLocalizedDescriptionKey: "swift view \(name) is not linked; rebuild the app"])
    }
  }
  static func build(_ held: AnyObject, _ name: String, _ emit: @escaping (String, String) -> Void) throws -> AnyView {
    switch name {
    case "Level":
      guard let props = held as? Held_Level else { throw NSError(domain: "OneNativeSource", code: 1, userInfo: [NSLocalizedDescriptionKey: "props of another view for \(name)"]) }
      return AnyView(Level(
        value: props.`value`,
        onChange: { (a0: Double) -> Void in emit("onChange", args([a0])) }
      ))
    default: throw NSError(domain: "OneNativeSource", code: 1, userInfo: [NSLocalizedDescriptionKey: "swift view \(name) is not linked; rebuild the app"])
    }
  }
  static func view(_ name: String, _ propsJson: String, _ emit: @escaping (String, String) -> Void) throws -> AnyView {
    guard let held = try update(nil, name, propsJson) else { throw NSError(domain: "OneNativeSource", code: 1, userInfo: [NSLocalizedDescriptionKey: "\(name) needs a full props document"]) }
    return try build(held, name, emit)
  }
