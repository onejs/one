`One.iOS` tab navigation and swipe paging are SwiftUI's own. `TabView` is a
controlled tab view of `Tab` pages, with `TabSection` groups for the iPad
sidebar, `TabViewSlot` and `TabViewBottomAccessory` content around the bar, and
an optional `Toolbar`. `Pager` is a tab view without the bar: people swipe
between `Page` children.

Reach for `TabView` for top-level navigation with a bar, and `Pager` for swipe
flows with no chrome. `selection` must name a mounted page tab, or both
throw. Every view here throws on web and Android.

## Usage

```tsx
import { useState } from 'react'
import { Text } from 'react-native'
import { One } from 'one'

function FeedScreen() {
  return <Text>Your feed.</Text>
}

function Library({ openComposer }: { openComposer: () => void }) {
  const [tab, setTab] = useState('feed')

  return (
    <One.iOS.TabView selection={tab} onSelectionChange={setTab}>
      <One.iOS.Tab value="feed" titleKey="Feed" systemImage="house">
        <FeedScreen />
      </One.iOS.Tab>
      <One.iOS.Tab value="new" titleKey="New" systemImage="plus" onPress={openComposer} />
    </One.iOS.TabView>
  )
}
```

The first tab shows a page; the second is an action tab that runs its press
and never becomes the selection. `onReselect` reports a tap on the selected
tab.

## How it works

Selection is a controlled tab value with an event count and a revision. A
native tab change bumps the count and reports the new id; JavaScript calls the
change handler and acknowledges the event. Raising `revision` with a new
selection moves the tab view from the outside. Action tabs and section header
buttons run their callbacks directly instead of changing the selection.

On iOS 18 and later each page renders in the modern `Tab` API inside its own
navigation container, which stays transparent so the system bar shows the
content behind it; your page owns its background. Older iOS falls back to a
legacy `TabView`. Sections, tab modifiers, styles, and customization need iOS
18. Tab customization edits persist as JSON: pass `onCustomizationChange` to
bind it, persist the reported string, and pass it back as `customization`.

## TabView

`TabView` is a SwiftUI tab view. Only `Tab`, `TabSection`, `TabViewSlot`,
`TabViewBottomAccessory`, and one `Toolbar` may be its direct children.

```tsx
import { useState } from 'react'
import { Text } from 'react-native'
import { One } from 'one'

function Home({ openComposer }: { openComposer: () => void }) {
  const [tab, setTab] = useState('feed')

  return (
    <One.iOS.TabView
      selection={tab}
      onSelectionChange={setTab}
      onReselect={(value) => console.log('reselected', value)}
    >
      <One.iOS.Tab value="feed" titleKey="Feed" systemImage="house">
        <Text>Your feed.</Text>
      </One.iOS.Tab>
      <One.iOS.Tab value="new" titleKey="New" systemImage="plus" onPress={openComposer} />
    </One.iOS.TabView>
  )
}
```

## Tab

`Tab` is one tab. It takes either children, as a page, or `onPress`, as an
action tab, never both. Values must be unique and nonempty, and `systemImage`
and `image` are mutually exclusive.

```tsx
import { Text } from 'react-native'
import { One } from 'one'

function SearchTab() {
  return (
    <One.iOS.Tab value="search" titleKey="Search" systemImage="star" role="search" badge={3}>
      <Text>Search results.</Text>
    </One.iOS.Tab>
  )
}
```

## TabSection

`TabSection` groups tabs under a heading in the sidebar. Its children are
`Tab` elements, at least one.

```tsx
import { useState } from 'react'
import { Text } from 'react-native'
import { One } from 'one'

function LibraryTabView() {
  const [tab, setTab] = useState('songs')

  return (
    <One.iOS.TabView tabViewStyle="sidebarAdaptable" selection={tab} onSelectionChange={setTab}>
      <One.iOS.TabSection id="music" title="Music">
        <One.iOS.Tab value="songs" titleKey="Songs" systemImage="star">
          <Text>Your songs.</Text>
        </One.iOS.Tab>
      </One.iOS.TabSection>
    </One.iOS.TabView>
  )
}
```

## TabViewSlot

`TabViewSlot` fills a named tab view slot with React Native content. Place it
directly inside `TabView`, alongside the `Tab` children. Each name may appear
once, the height must be above zero, and the slot keeps its declared height
and React Native event handlers as selection changes.

```tsx
import { useState } from 'react'
import { Pressable, Text } from 'react-native'
import { One } from 'one'

function PlayerTabView({ openPlayer }: { openPlayer: () => void }) {
  const [tab, setTab] = useState('home')

  return (
    <One.iOS.TabView selection={tab} onSelectionChange={setTab}>
      <One.iOS.Tab value="home" titleKey="Home" systemImage="house">
        <Text>Home.</Text>
      </One.iOS.Tab>
      <One.iOS.Tab value="other" titleKey="Other" systemImage="star">
        <Text>Other.</Text>
      </One.iOS.Tab>
      <One.iOS.TabViewSlot name="tabViewBottomAccessory" height={50}>
        <Pressable onPress={openPlayer}>
          <Text>Open player</Text>
        </Pressable>
      </One.iOS.TabViewSlot>
    </One.iOS.TabView>
  )
}
```

The bottom accessory remains above the system tab bar as selection changes.
The sidebar slots are `tabViewSidebarHeader`, `tabViewSidebarFooter`, and
`tabViewSidebarBottomBar`: they display only when the tab view presents a
sidebar and are absent from the compact iPhone tab bar. On iPad, use
`tabViewStyle="sidebarAdaptable"` to present the native sidebar; the header
and footer sit above the sidebar bottom bar, which spans the bottom of the
sidebar.

```tsx
import { useState } from 'react'
import { Pressable, Text } from 'react-native'
import { One } from 'one'

function SidebarTabView({ openSearch, openSettings, openProfile }: { openSearch: () => void; openSettings: () => void; openProfile: () => void }) {
  const [tab, setTab] = useState('home')

  return (
    <One.iOS.TabView tabViewStyle="sidebarAdaptable" selection={tab} onSelectionChange={setTab}>
      <One.iOS.Tab value="home" titleKey="Home" systemImage="house">
        <Text>Home.</Text>
      </One.iOS.Tab>
      <One.iOS.Tab value="other" titleKey="Other" systemImage="star">
        <Text>Other.</Text>
      </One.iOS.Tab>
      <One.iOS.TabViewSlot name="tabViewSidebarHeader" height={44}>
        <Pressable onPress={openSearch}>
          <Text>Search</Text>
        </Pressable>
      </One.iOS.TabViewSlot>
      <One.iOS.TabViewSlot name="tabViewSidebarFooter" height={48}>
        <Pressable onPress={openSettings}>
          <Text>Settings</Text>
        </Pressable>
      </One.iOS.TabViewSlot>
      <One.iOS.TabViewSlot name="tabViewSidebarBottomBar" height={52}>
        <Pressable onPress={openProfile}>

}
```

## TabViewBottomAccessory

`TabViewBottomAccessory` sits above the tab bar, like a music player's now
playing bar. It needs iOS 26, and `TabView` takes one. It needs `children`,
`inline`, or `expanded`; a missing layout falls back to the others.

```tsx
import { useState } from 'react'
import { Text } from 'react-native'
import { One } from 'one'

function PlayerBar() {
  const [tab, setTab] = useState('home')

  return (
    <One.iOS.TabView selection={tab} onSelectionChange={setTab}>
      <One.iOS.Tab value="home" titleKey="Home" systemImage="house">
        <Text>Home.</Text>
      </One.iOS.Tab>
      <One.iOS.TabViewBottomAccessory
        inline={<Text>Evening Tide</Text>}
        expanded={<Text>Evening Tide, The Harbor Lights</Text>}
      />
    </One.iOS.TabView>
  )
}
```

## Pager, Page

`Pager` is a tab view without the bar: people swipe between pages. Children
must be `Page` elements with unique nonempty ids, and `selection` must name a
mounted page. For a pager on both platforms, use
[One.UI.Pager](/native/ui-pager).

Use `indexDisplayMode="always"` to show page indicators, `"never"` to hide them,
or `"automatic"` to let SwiftUI choose. The default is `"automatic"`.

```tsx
import { useState } from 'react'
import { Text } from 'react-native'
import { One } from 'one'

function Onboarding() {
  const [page, setPage] = useState('one')

  return (
    <One.iOS.Pager selection={page} onSelectionChange={setPage}>
      <One.iOS.Page id="one">
        <Text>Welcome.</Text>
      </One.iOS.Page>
      <One.iOS.Page id="two">
        <Text>How it works.</Text>
      </One.iOS.Page>
    </One.iOS.Pager>
  )
}
```

## Platform differences

| | iOS | Android | Web |
| --- | --- | --- | --- |
| Renders | SwiftUI tabs | Throws | Throws |
| Sidebar | iPad sidebar | None | None |
| Paging | Swipe pages | None | None |

In the Peach simulator `TabView` draws the floating tab bar and honours
`swiftStyle={{ tabBarMinimizeBehavior }}`: the bar collapses as the page
scrolls down, and a tap on it expands it again. The `page` tab view style and
the iPad sidebar are not drawn there, so Peach throws for them.

## Limits and gotchas

- **`tabBarMinimizeBehavior` follows the page's first scroll view.** One names a
  page's first React Native `ScrollView` (or list) as the tab bar's content
  scroll view, so the bar minimizes as it scrolls with no extra wiring.
- **Selection must name a mounted page.** Both `TabView` and `Pager` throw
  otherwise, and an action tab never becomes the selection.
- **TabView takes markers only.** Anything but `Tab`, `TabSection`,
  `TabViewSlot`, `TabViewBottomAccessory`, and one `Toolbar` as direct
  children throws; markers rendered anywhere else throw too.
- **A tab is a page or an action.** Exactly one of children and `onPress`,
  and exactly one of `systemImage` and `image`.
- **New chrome needs new iOS.** Sections, modifiers, styles, and
  customization need iOS 18; the bottom accessory needs iOS 26, its
  `isEnabled` iOS 26.1, and tab help text iOS 27.
